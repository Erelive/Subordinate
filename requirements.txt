# Audio capture (WASAPI loopback fork of PyAudio)
PyAudioWPatch==0.2.12.7

# ASR: CTranslate2 backend, bundles a Silero VAD ONNX model
faster-whisper==1.2.0
# faster-whisper 1.2.0 imports requests but does not declare it
requests==2.32.5

# CUDA runtime libraries CTranslate2 links against on Windows.
# Without these, GPU inference fails with "Library cublas64_12.dll is not found".
nvidia-cublas-cu12==12.6.4.1
nvidia-cudnn-cu12==9.6.0.74

# High-quality resampling, 48 kHz device rate -> 16 kHz model rate
soxr==0.5.0.post1

# Tokenizer for the NMT translation stage (subordinant/translate.py). The model
# itself runs on CTranslate2, which faster-whisper already pulls in, so this is
# the only extra dependency translation costs.
sentencepiece==0.2.2

# Speaker embeddings for the labelling stage (subordinant/diarize.py). Chosen
# over pyannote.audio because that needs PyTorch (~2.5 GB) and a gated model
# download, either of which would end the double-click-to-run story.
sherpa-onnx==1.13.6
# Pulled in by faster-whisper for its Silero VAD, but pinned here because
# sherpa-onnx binds to this DLL directly. Windows 11 ships its own, older
# onnxruntime.dll in System32, and sherpa-onnx segfaults against it - see
# cuda.preload_onnxruntime for how the right copy is pinned.
onnxruntime==1.29.0

numpy==2.2.1

# Overlay window
PyQt6==6.8.0
